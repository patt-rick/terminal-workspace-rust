# What's New for New Versions — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a curated `CHANGELOG.md` that is the single source of release notes, surfaced as an improved pre-install update dialog, a post-install "What's New" recap, and an anytime in-app release history — with a release script that makes forgetting the entry nearly impossible.

**Architecture:** `CHANGELOG.md` (Keep-a-Changelog) is bundled into the React app via Vite `?raw` and parsed by a pure TS module; the same notes feed the GitHub release body (via a self-contained Node extractor in CI) so the updater's pre-install dialog shows them too. A `scripts/release.mjs` stamps the changelog + bumps the four version files atomically and refuses if the `Unreleased` section is empty.

**Tech Stack:** React 19 + TypeScript, Zustand, Vite 6, Vitest, Tauri 2 (`@tauri-apps/plugin-updater`), `react-markdown` + `remark-gfm`, GitHub Actions + `tauri-apps/tauri-action`.

Spec: `docs/superpowers/specs/2026-09-14-whats-new-design.md`

```ts
export interface ChangelogEntry {
  /** `x.y.z` version, no leading `v` */
  version: string
  /** ISO-ish date string from the heading, or null if absent */
  date: string | null
  /** markdown body between this heading and the next `## [` heading, trimmed */
  body: string
}

const HEADING = /^## \[([^\]]+)\](?:\s*-\s*(.+?))?\s*$/

/** Parse CHANGELOG.md text into released entries (newest-first), skipping Unreleased. */
export function parseChangelog(text: string): ChangelogEntry[] {
  const lines = text.split(/\r?\n/)
  const entries: ChangelogEntry[] = []
  let current: { version: string; date: string | null; body: string[] } | null = null

  const flush = (): void => {
    if (current && current.version.toLowerCase() !== 'unreleased') {
      entries.push({
        version: current.version,
        date: current.date,
        body: current.body.join('\n').trim(),
      })
    }
  }

  for (const line of lines) {
    const m = HEADING.exec(line)
    if (m) {
      flush()
      current = { version: m[1].trim(), date: m[2]?.trim() ?? null, body: [] }
    } else if (current) {
      current.body.push(line)
    }
  }
  flush()
  return entries
}

/** Numeric field-by-field compare of two `x.y.z` versions. Malformed fields count as 0. */
export function compareVersions(a: string, b: string): number {
  const parse = (v: string): number[] => v.split('.').map((n) => parseInt(n, 10))
  const pa = parse(a)
  const pb = parse(b)
  for (let i = 0; i < 3; i += 1) {
    const x = Number.isFinite(pa[i]) ? pa[i] : 0
    const y = Number.isFinite(pb[i]) ? pb[i] : 0
    if (x !== y) return x < y ? -1 : 1
  }
  return 0
}

/** Entries whose version is strictly newer than `sinceVersion`, newest-first. */
export function entriesNewerThan(
  entries: ChangelogEntry[],
  sinceVersion: string,
): ChangelogEntry[] {
  return entries.filter((e) => compareVersions(e.version, sinceVersion) > 0)
}
```

```js
import { readFileSync } from 'node:fs'
import { fileURLToPath } from 'node:url'

/**
 * Extract the CHANGELOG.md section body for a tag/version. A leading `v` is
 * stripped (git tags are `v0.4.2`; headings are `## [0.4.2]`). The heading line
 * itself is dropped so the notes read cleanly. Unknown version -> ''.
 */
export function sectionFor(changelog, tag) {
  const version = String(tag ?? '').replace(/^v/, '').trim()
  if (!version) return ''
  const lines = changelog.split(/\r?\n/)
  const heading = /^## \[([^\]]+)\]/
  const out = []
  let capturing = false
  for (const line of lines) {
    const m = heading.exec(line)
    if (m) {
      if (capturing) break // reached the next section
      if (m[1].trim() === version) {
        capturing = true // start after this heading; do not push it
      }
      continue
    }
    if (capturing) out.push(line)
  }
  return out.join('\n').trim()
}

// CLI: `node scripts/changelog-section.mjs <tag>` -> prints the section to stdout.
const invokedDirectly =
  process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1]
if (invokedDirectly) {
  const tag = process.argv[2] ?? ''
  let text = ''
  try {
    text = readFileSync(new URL('../CHANGELOG.md', import.meta.url), 'utf8')
  } catch {
    // no changelog -> empty notes, exit 0
  }
  process.stdout.write(sectionFor(text, tag))
}
```

**Files:**

- Modify: `src/app.tsx:407`

- [ ] **Step 1: Implement the dialog**

```tsx
import { useEffect, useState } from 'react'
import { isTauri } from '../lib/ipc'
import { useUpdater } from '../state/updater'
import {
  parseChangelog,
  compareVersions,
  entriesNewerThan,
  type ChangelogEntry,
} from '../lib/changelog'
import { ChangelogList } from './changelog-list'
import changelogRaw from '../../CHANGELOG.md?raw'

const LAST_SEEN_KEY = 'tw:last-seen-version'

function readLastSeen(): string | null {
  try {
    return localStorage.getItem(LAST_SEEN_KEY)
  } catch {
    return null
  }
}

function stampSeen(version: string): void {
  try {
    localStorage.setItem(LAST_SEEN_KEY, version)
  } catch {
    // storage may be unavailable — nothing else to do
  }
}

/**
 * Mounted once at the app root, next to UpdateManager. On first launch of a
 * newer version (vs. the stored last-seen), shows the changelog entries added
 * since. First-ever launch / fresh install / cleared storage stamps silently.
 * Only runs inside Tauri, and defers while an update dialog is showing.
 */
export function WhatsNewDialog() {
  const [entries, setEntries] = useState<ChangelogEntry[] | null>(null)
  const updaterStatus = useUpdater((s) => s.status)

  useEffect(() => {
    if (!isTauri) return
    const current = __APP_VERSION__
    const last = readLastSeen()
    if (!last) {
      stampSeen(current)
      return
    }
    if (compareVersions(current, last) <= 0) {
      stampSeen(current) // same version or a downgrade — nothing to show
      return
    }
    const newer = entriesNewerThan(parseChangelog(changelogRaw), last)
    if (newer.length === 0) {
      stampSeen(current) // version moved but changelog has no matching entries
      return
    }
    setEntries(newer)
  }, [])

  // Don't stack over the updater's own dialog. Subscribed to status, so if an
  // update surfaces after mount we hide, and reveal again once it's dismissed.
  const updaterBusy =
    updaterStatus === 'checking' ||
    updaterStatus === 'available' ||
    updaterStatus === 'downloading' ||
    updaterStatus === 'error'

  if (!entries || updaterBusy) return null

  const dismiss = (): void => {
    stampSeen(__APP_VERSION__)
    setEntries(null)
  }

  return (
    <div
      className="fixed inset-0 z-50 flex items-center justify-center"
      style={{ background: 'var(--backdrop)' }}
      onClick={dismiss}
    >
      <div
        className="flex max-h-[80vh] w-[440px] flex-col overflow-hidden rounded-xl border border-border bg-overlay p-5 shadow-xl"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="text-sm font-semibold text-foreground">
          What's New in v{__APP_VERSION__}
        </div>
        <div className="mt-3 overflow-auto">
          <ChangelogList entries={entries} />
        </div>
        <div className="mt-5 flex justify-end">
          <button
            type="button"
            onClick={dismiss}
            className="rounded-md bg-accent px-3 py-1.5 text-sm font-medium text-accent-foreground hover:opacity-90"
          >
            Got it
          </button>
        </div>
      </div>
    </div>
  )
}
```

```js
import { readFileSync, writeFileSync } from 'node:fs'
import { execFileSync } from 'node:child_process'

/** Rename `## [Unreleased]` -> `## [version] - date` and open a fresh empty
 *  Unreleased above it. Throws if Unreleased is missing or empty. */
export function stampUnreleased(changelog, version, date) {
  const lines = changelog.split(/\r?\n/)
  const idx = lines.findIndex((l) => /^## \[Unreleased\]/i.test(l))
  if (idx === -1) throw new Error('No "## [Unreleased]" section found in CHANGELOG.md')

  let end = lines.length
  for (let i = idx + 1; i < lines.length; i += 1) {
    if (/^## \[/.test(lines[i])) {
      end = i
      break
    }
  }
  const body = lines.slice(idx + 1, end).join('\n').trim()
  if (!body) {
    throw new Error(
      'The "## [Unreleased]" section is empty — add release notes before cutting a release.',
    )
  }

  const rebuilt = [
    ...lines.slice(0, idx),
    '## [Unreleased]',
    '',
    `## [${version}] - ${date}`,
    ...lines.slice(idx + 1),
  ]
  return rebuilt.join('\n')
}

/** Replace the first `"version": "..."` key (package version) in a JSON file. */
export function bumpJsonVersion(text, version) {
  return text.replace(/("version"\s*:\s*")[^"]+(")/, `$1${version}$2`)
}

/** Replace the `version = "..."` under `[package]` in Cargo.toml. */
export function bumpCargoToml(text, version) {
  return text.replace(/(\[package\][\s\S]*?\nversion\s*=\s*")[^"]+(")/, `$1${version}$2`)
}

/** Replace the version of the `terminal-workspace` package in Cargo.lock. */
export function bumpCargoLock(text, version) {
  return text.replace(
    /(name = "terminal-workspace"\nversion = ")[^"]+(")/,
    `$1${version}$2`,
  )
}

// CLI: `node scripts/release.mjs <x.y.z>`
import { fileURLToPath } from 'node:url'
const invokedDirectly =
  process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1]
if (invokedDirectly) {
  const version = process.argv[2]
  if (!version || !/^\d+\.\d+\.\d+$/.test(version)) {
    console.error('Usage: node scripts/release.mjs <x.y.z>')
    process.exit(1)
  }
  const date = new Date().toISOString().slice(0, 10)
  const root = new URL('../', import.meta.url)
  const at = (rel) => new URL(rel, root)

  const read = (u) => readFileSync(u, 'utf8')

  writeFileSync(at('CHANGELOG.md'), stampUnreleased(read(at('CHANGELOG.md')), version, date))
  writeFileSync(at('package.json'), bumpJsonVersion(read(at('package.json')), version))
  writeFileSync(at('src-tauri/tauri.conf.json'), bumpJsonVersion(read(at('src-tauri/tauri.conf.json')), version))
  writeFileSync(at('src-tauri/Cargo.toml'), bumpCargoToml(read(at('src-tauri/Cargo.toml')), version))
  writeFileSync(at('src-tauri/Cargo.lock'), bumpCargoLock(read(at('src-tauri/Cargo.lock')), version))

  const files = [
    'CHANGELOG.md',
    'package.json',
    'src-tauri/tauri.conf.json',
    'src-tauri/Cargo.toml',
    'src-tauri/Cargo.lock',
  ]
  execFileSync('git', ['add', ...files], { stdio: 'inherit' })
  execFileSync('git', ['commit', '-m', `chore(release): bump to ${version}`], { stdio: 'inherit' })

  console.log(`\nReleased ${version}. Next:\n  git tag v${version} && git push && git push --tags`)
```
